        "# 3. Company Classification (Order matters!)\n",
        "def classify_company(name):\n",
        "    if not isinstance(name, str):\n",
        "        return \"Other\"\n",
        "\n",
        "    # Exact word/pattern checks as specified:\n",
        "    if re.search(r'\\bTechnologies\\b', name):\n",
        "        return \"Technologies\"\n",
        "    elif re.search(r'\\bAcquisition Corp\\b|\\bAcquisition Corporation\\b|\\bCorp\\b', name):\n",
        "        return \"Acquisition Corp\"\n",
        "    elif re.search(r'\\bInc\\b|\\bIncorporated\\b', name):\n",
        "        return \"Inc.\"\n",
        "    elif re.search(r'\\bGroup\\b', name):\n",
        "        return \"Group\"\n",
        "    elif re.search(r'\\bLtd\\b|\\bLimited\\b', name):\n",
        "        return \"Limited\"\n",
        "    elif re.search(r'\\bHoldings\\b|\\bHolding\\b', name):\n",
        "        return \"Holdings\"\n",

        "df_withdrawn['Company Type'] = df_withdrawn['Company'].apply(classify_company)\n",
        "\n",
        "# 4. Price Parsing\n",
        "def parse_price(val):\n",
        "    if pd.isna(val) or str(val).strip() in ['-', '']:\n",
        "        return np.nan\n",
        "    cleaned = re.sub(r'[^\\d.]', '', str(val))\n",
        "    try:\n",
        "        return float(cleaned)\n",
        "    except ValueError:\n",
        "        return np.nan\n",
        "\n",
        "df_withdrawn['Price Low Num'] = df_withdrawn['Price Low'].apply(parse_price)\n",
        "df_withdrawn['Price High Num'] = df_withdrawn['Price High'].apply(parse_price)\n",

        "\n",
        "# 5. Numeric Conversion for Shares and Est $ Vol\n",
        "def parse_numeric(val):\n",
        "    if pd.isna(val) or str(val).strip() in ['-', '']:\n",
        "        return np.nan\n",
        "    cleaned = re.sub(r'[^\\d.]', '', str(val))\n",
        "    try:\n",
        "        return float(cleaned)\n",
        "    except ValueError:\n",
        "        return np.nan\n",

        "df_withdrawn['Shares_num'] = df_withdrawn['Shares (millions)'].apply(parse_numeric)\n",
